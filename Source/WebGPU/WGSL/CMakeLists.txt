# Debug builds incrementally in batch mode, the other configurations whole-module. The driver warns
# about WebKitSwiftFlags.cmake's -enable-batch-mode together with CMake's -num-threads, and ignores
# -num-threads in batch mode and -enable-batch-mode under -wmo, so keep only the one that applies.
if (CMAKE_BUILD_TYPE STREQUAL "Debug")
    string(REPLACE " -num-threads ${CMAKE_Swift_NUM_THREADS}" "" CMAKE_Swift_COMPILE_OBJECT "${CMAKE_Swift_COMPILE_OBJECT}")
    string(FIND "${CMAKE_Swift_COMPILE_OBJECT}" "-num-threads" _num_threads)
    if (NOT _num_threads EQUAL -1)
        message(WARNING "WGSL: -num-threads is still in CMAKE_Swift_COMPILE_OBJECT: ${CMAKE_Swift_COMPILE_OBJECT}")
    endif ()
endif ()

function(WGSL_SETUP_SWIFT_TARGET _target)
    set_target_properties(${_target} PROPERTIES
        Swift_COMPILATION_MODE "$<IF:$<CONFIG:Debug>,incremental,wholemodule>"
    )
    set(_debug_batch_mode "$<$<AND:$<COMPILE_LANGUAGE:Swift>,$<CONFIG:Debug>>:-enable-batch-mode>")
    get_target_property(_options ${_target} COMPILE_OPTIONS)
    string(REPLACE "$<$<COMPILE_LANGUAGE:Swift>:-enable-batch-mode>" "${_debug_batch_mode}" _options "${_options}")
    string(REPLACE "${_debug_batch_mode}" "" _other_options "${_options}")
    string(FIND "${_other_options}" "-enable-batch-mode" _batch_mode)
    if (NOT _batch_mode EQUAL -1)
        message(FATAL_ERROR "${_target}: -enable-batch-mode would reach the -wmo configurations; match WebKitSwiftFlags.cmake's spelling of it here")
    endif ()
    set_target_properties(${_target} PROPERTIES COMPILE_OPTIONS "${_options}")
endfunction()

set_target_properties(WGSL PROPERTIES
    Swift_MODULE_NAME WGSL
)
WGSL_SETUP_SWIFT_TARGET(WGSL)
webkit_target_add_swift_options(WGSL
    "$<$<AND:$<COMPILE_LANGUAGE:Swift>,$<CONFIG:Debug>>:-DWGSL_UNOPTIMIZED>"
)
WEBKIT_RAISE_SWIFT_NINJA_PRIORITY(WGSL)

# wgslc executable
set(wgslc_SOURCES
    wgslc/WGSLBench.swift
    wgslc/wgslc.swift
)
set(wgslc_LIBRARIES WGSL)
WEBKIT_EXECUTABLE_DECLARE(wgslc)
# Before WEBKIT_EXECUTABLE(), which reads the bundle properties.
if (WEBKIT_SDK_IS_IOS_FAMILY)
    set_target_properties(wgslc PROPERTIES MACOSX_BUNDLE FALSE)
endif ()
WEBKIT_EXECUTABLE(wgslc)
WGSL_SETUP_SWIFT_TARGET(wgslc)
# @main outside a file named main.swift.
webkit_target_add_swift_options(wgslc
    -parse-as-library
)
# Without -force_load the linker drops archive members that hold only types and conformances.
target_link_options(wgslc PRIVATE "LINKER:-force_load,$<TARGET_FILE:WGSL>")

    RUNTIME DESTINATION "${LIBEXEC_INSTALL_DIR}"
    BUNDLE DESTINATION "${LIBEXEC_INSTALL_DIR}")
